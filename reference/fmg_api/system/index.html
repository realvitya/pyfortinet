@field_validator("secondaryip")
def standardize_secondaryip(cls, value):
    """Accept schema subtable rows as well as the legacy address shorthand."""
    if value is None:
        return value
    if isinstance(value, list) and all(isinstance(row, _system_interface.InterfaceSecondaryip) for row in value):
        return value
    return cls.standardize_subnet(value)

@field_validator("ip", "management_ip")
def standardize_subnet(cls, v):
    """validator: x.x.x.x/y.y.y.y -> x.x.x.x/y

    API use this list form: ["1.2.3.4", "255.255.255.0"]
    Human use this form: "1.2.3.4/24"
    """
    if v is None:
        return None
    if isinstance(v, list):
        return IPv4Interface("/".join(v)).compressed
    else:
        return IPv4Interface(v).compressed

@field_serializer("vdom", when_used="unless-none")
def serialize_vdom(self, value: Union[str, VDOM, list[Union[str, VDOM]]]) -> list[str]:
    """Serialize the schema's VDOM name array, including object-reference inputs."""
    values = value if isinstance(value, list) else [value]
    return [item.name if isinstance(item, VDOM) else item for item in values]
